\vec{n}^{\top}\vec{r}\brak{\theta} &= \myvec{\cos\theta_0 & \sin\theta_0}\myvec{\cos\theta \\ \sin\theta} = \cos\brak{\theta - \theta_0}

\vec{u}^{\top}\vec{r}\brak{\theta} &= -p e^2 \cos\brak{\theta - \theta_0}
\end{align}
Substituting these into \eqref{eq:EE26BTECH11202-polar_quad}:
\begin{align}
r^2 \sbrak{1 - e^2 \cos^2\brak{\theta - \theta_0}} - 2r p e^2 \cos\brak{\theta - \theta_0} - p^2 e^2 = 0

\item Specific angular momentum for a satellite with position vector $\vec{\vec{r}}$ and velocity vector $\vec{\vec{v}}$ is 
\begin{align}
	h= \norm{\vec{r} \times \vec{v}} \label{eq:EE26BTECH11202-ang-moment}
\end{align}
\item Specific Energy for a satellite with position vector $\vec{\vec{r}}$, velocity vector $\vec{\vec{v}}$ and gravitational parameter $\mu$ is 
\begin{align}
\varepsilon = \frac{v^2}{2} - \frac{\mu}{r} \label{eq:EE26BTECH11202-energy}
\end{align}
where
\begin{align}
	r=\norm{\vec{r}}, v=\norm{\vec{v}}
\end{align}
\item The gravitational acceleration $\vec{a}$ acting on a satellite is given by Newton's law of gravitation:
\begin{align}
\vec{a} = -\frac{\mu}{r^3}\vec{r}
\label{eq:EE26BTECH11202-grav_acc}
\end{align}
\item The trajectory of a satellite under a central gravitational force is always a conic section. \\
\solution
Taking the cross product of $\vec{r}$ with $\vec{a}$:
\begin{align}
	\vec{r} \times \vec{a} = \vec{r} \times \brak{-\frac{\mu}{r^3}\vec{r}} = \vec{0} \label{eq:EE26BTECH11202-grav-intermediate}
\end{align}
From \eqref{eq:EE26BTECH11202-grav-intermediate}
\begin{align}\frac{d}{dt}\brak{\vec{r} \times \vec{v}} = \vec{v} \times \vec{v} + \vec{r} \times \vec{a} = \vec{0}
\end{align}
Hence, the specific angular momentum vector $\vec{h}$ is conserved:
\begin{align}
\vec{h} = \vec{r} \times \vec{v} = \text{constant}
\label{eq:EE26BTECH11202-ang_mom}
\end{align}

Crossing the acceleration $\vec{a}$ with $\vec{h}$:
\begin{align}
\vec{a} \times \vec{h} &= \brak{-\frac{\mu}{r^3}\vec{r}} \times \brak{\vec{r} \times \vec{v}} 
\\
&= -\frac{\mu}{r^3} \sbrak{\brak{\vec{r} \cdot \vec{v}}\vec{r} - \brak{\vec{r} \cdot \vec{r}}\vec{v}} 
\\
&= \frac{\mu}{r^3} \sbrak{r^2\vec{v} - \brak{\vec{r} \cdot \vec{v}}\vec{r}}
\end{align}
Using $\vec{r} \cdot \vec{v} = r \frac{dr}{dt}$, this simplifies to:
\begin{align}
\vec{a} \times \vec{h} = \mu \sbrak{\frac{\vec{v}}{r} - \frac{1}{r^2}\frac{dr}{dt}\vec{r}} = \mu \frac{d}{dt}\brak{\frac{\vec{r}}{r}}
\end{align}
Integrating with respect to time $t$:
\begin{align}
\vec{v} \times \vec{h} = \mu \brak{\frac{\vec{r}}{r} + \vec{C}}
\label{eq:EE26BTECH11202-v_cross_h}
\end{align}
where $\vec{C}$ is the integration constant. Taking the scalar product of \eqref{eq:EE26BTECH11202-v_cross_h} with $\vec{r}$:
\begin{align}
\vec{r} \cdot \brak{\vec{v} \times \vec{h}} &= \mu \vec{r} \cdot \brak{\frac{\vec{r}}{r} + \vec{C}}
\\
\implies \brak{\vec{r} \times \vec{v}} \cdot \vec{h} &= \mu \brak{r + r C \cos\theta}
\\
\implies \norm{\vec{h}}^2 &= \mu r \brak{1 + C \cos\theta}
\end{align}
where $\theta$ is the angle between $\vec{r}$ and $\vec{C}$. Rearranging for $r$ yields 
\begin{align}
	r = \frac{\frac{\norm{\vec{h}}^2}{\mu}}{1 + C \cos\theta} = \frac{ep}{1 + e \cos\theta}
\end{align}
This is the same equation as \eqref{eq:EE26BTECH11202-polar_conic} where
\begin{align}
ep = \frac{\norm{\vec{h}}^2}{\mu}
\end{align}
\begin{align}
	\vec{e}=\vec{C} \label{eq:EE26BTECH11202-eccentricity_vector}
\end{align}
\item The squared eccentricity $e^2$ of a satellite's trajectory is related to its specific orbital energy $\varepsilon$ and specific angular momentum magnitude $h$ by
\begin{align}
	e^2 = 1 + \frac{2\varepsilon h^2}{\mu^2} \label{eq:EE26BTECH11202-eccentricity}
\end{align}
\solution
From \eqref{eq:EE26BTECH11202-v_cross_h} and \eqref{eq:EE26BTECH11202-eccentricity_vector},
\begin{align}
\vec{e} = \frac{\vec{v} \times \vec{h}}{\mu} - \frac{\vec{r}}{r}
\end{align}
Taking the magnitude squared of both sides using $\norm{\vec{e}}^2 = \vec{e} \cdot \vec{e}$:
\begin{align}
e^2 = \norm{\vec{e}}^2 = \brak{\frac{\vec{v} \times \vec{h}}{\mu} - \frac{\vec{r}}{r}} \cdot \brak{\frac{\vec{v} \times \vec{h}}{\mu} - \frac{\vec{r}}{r}}
\end{align}
Expanding the inner product:
\begin{align}
e^2 = \frac{\norm{\vec{v} \times \vec{h}}^2}{\mu^2} - \frac{2}{\mu r} \sbrak{\vec{r} \cdot \brak{\vec{v} \times \vec{h}}} + \frac{\vec{r} \cdot \vec{r}}{r^2}
\label{eq:EE26BTECH11202-ecc_expanded}
\end{align}

We know that
\begin{align}
\vec{r} \cdot \brak{\vec{v} \times \vec{h}} = \brak{\vec{r} \times \vec{v}} \cdot \vec{h} = \vec{h} \cdot \vec{h} = h^2
\end{align}
And,
\begin{align}
\vec{r} \cdot \vec{r} = r^2
\end{align}
Substituting these back into \eqref{eq:EE26BTECH11202-ecc_expanded} yields
\begin{align}
e^2 &= \frac{v^2 h^2}{\mu^2} - \frac{2 h^2}{\mu r} + 1 
\\
&= 1 + \frac{2 h^2}{\mu^2} \brak{\frac{v^2}{2} - \frac{\mu}{r}}
\label{eq:EE26BTECH11202-ecc_energy_sub}
\end{align}

Substituting \eqref{eq:EE26BTECH11202-energy} into \eqref{eq:EE26BTECH11202-ecc_energy_sub} yields:
\begin{align}
e^2 = 1 + \frac{2\varepsilon h^2}{\mu^2}
\end{align}
