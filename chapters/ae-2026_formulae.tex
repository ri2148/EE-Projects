\end{align}
\begin{align}
\implies \|\vec{n}\|^2 (\vec{x} - \vec{F})^\top (\vec{x} - \vec{F}) &= e^2 \left(\vec{n}^\top \vec{x} - c\right)^2

\end{align}
\begin{align}

\item Results on Gravitation
\begin{itemize}
\item Specific angular momentum - Consider a satellite with position vector $\vec{\vec{r}}$ and velocity vector $\vec{\vec{v}}$
\begin{align}
	h_0 = |\vec{r} \times \vec{v}| \label{eq:EE26BTECH11202-Gravitation_1}
\end{align}
\item Eccentricity of trajectory
\begin{align}
	e^2 = 1 + \frac{2\varepsilon h_0^2}{\mu^2} \label{eq:EE26BTECH11202-Gravitation_2}
\end{align}

\item Trajectory of a satellite given its position vector, velocity vector, and gravitational parameter $\mu$: The polar trajectory of a satellite with eccentricity vector $\vec{e}$ and semi-latus rectum $p_{\text{orb}}$ is given by:
\begin{align}
	r = \frac{p_{\text{orb}}}{1 + e \cos \theta} \label{eq:EE26BTECH11202-traj_1}
\end{align}

In vector form: 
\begin{align}
	e \cos \theta = \frac{\vec{e}^\top \vec{x}}{r} \label{eq:EE26BTECH11202-traj_2}
\end{align}
Where: 
\begin{align}
\vec{x} = \myvec{r\cos{\theta} \\ r\sin{\theta}}.
\end{align}
From \eqref{eq:EE26BTECH11202-traj_1} and \eqref{eq:EE26BTECH11202-traj_2}, we have:
\begin{align}
r + \vec{e}^\top \vec{x} = p_{\text{orb}} \implies r = p_{\text{orb}} - \vec{e}^\top \vec{x}
\end{align}

Squaring both sides :
\begin{align}
r^2 = \vec{x}^\top \vec{x} = (p_{\text{orb}} - \vec{e}^\top \vec{x})^2 = p_{\text{orb}}^2 - 2 p_{\text{orb}} \vec{e}^\top \vec{x} + (\vec{e}^\top \vec{x})^2
\end{align}

Rearranging yields:
\begin{align}
\vec{x}^\top (\vec{I} - \vec{e}\vec{e}^\top)\vec{x} + 2(p_{\text{orb}}\vec{e})^\top \vec{x} - p_{\text{orb}}^2 = 0
\end{align}

Comparing this with \eqref{eq:EE26BTECH11202-app-conic_quad_form}:
\begin{itemize}
    \item Matrix $\vec{V}$: $\vec{V} = \vec{I} - \vec{e}\vec{e}^\top$
    \item Vector $\vec{u}$: $\vec{u} = p_{\text{orb}}\vec{e}$
    \item Scalar $f$: $f = -p_{\text{orb}}^2$
\end{itemize}
From result \eqref{eq:EE26BTECH11202-mat_det_lemma}, we have:
\begin{align}
	\mydet{\vec{V}}=1-\vec{e^{\top}e}=1-e^2 \label{eq:EE26BTECH11202-Gravitation_3}
\end{align}
From the sign of $\mydet{\vec{V}}$, we get the trajectory.

\end{itemize}
\end{enumerate}
