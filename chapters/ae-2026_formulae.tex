\item
The equation of  a conic with directrix $\vec{n}^{\top}\vec{x} = c$, eccentricity $e$ and focus $\vec{F}$ is given by 
\begin{align}
    \label{eq:202-app-conic_quad_form}
	\text{g}\brak{\vec{x}} = \vec{x}^{\top}\vec{V}\vec{x}+2\vec{u}^{\top}\vec{x}+f=0
    \end{align}
where     
\begin{align}
  \label{eq:202-app-conic_quad_form_v}
\vec{V} &=\norm{\vec{n}}^2\vec{I}-e^2\vec{n}\vec{n}^{\top}, 
\\
\label{eq:202-app-conic_quad_form_u}
\vec{u} &= ce^2\vec{n}-\norm{\vec{n}}^2\vec{F}, 
\\
\label{eq:202-app-conic_quad_form_f}
f &= \norm{\vec{n}}^2\norm{\vec{F}}^2-c^2e^2
    \end{align}
\solution

\begin{align}
\|\vec{x} - \vec{F}\|^2 &= e^2 \frac{(\vec{n}^\top \vec{x} - c)^2}{\|\vec{n}\|^2} 
\end{align}
\begin{align}
\implies \|\vec{n}\|^2 (\vec{x} - \vec{F})^\top (\vec{x} - \vec{F}) &= e^2 \left(\vec{n}^\top \vec{x} - c\right)^2 
\end{align}
\begin{align}
\implies \|\vec{n}\|^2 \left(\vec{x}^\top \vec{x} - 2\vec{F}^\top \vec{x} + \|\vec{F}\|^2\right) &= e^2 \left(c^2 + \left(\vec{n}^\top \vec{x}\right)^2 - 2c\vec{n}^\top \vec{x}\right) 
\end{align}
\begin{align}
= e^2 \left(c^2 + \left(\vec{x}^\top \vec{n}\vec{n}^\top \vec{x}\right) - 2c\vec{n}^\top \vec{x}\right)
\end{align}
which can be expressed as \eqref{eq:202-app-conic_quad_form} after simplification.
\item
  The eccentricity of \eqref{eq:202-app-conic_quad_form} is given by 
\begin{align}
  \label{eq:202-app-conic_quad_form_e} 
  e&= \sqrt{1-\frac{\lambda_1}{\lambda_2}}
\end{align} 
Here, $\lambda_1$ and $\lambda_2$ are the eigenvalues of $\vec{V}$ 
	\label{app:conic-parameters}

\solution

From \eqref{eq:202-app-conic_quad_form_v}, using the fact that $\vec{V}$ is symmetric, 
  \begin{align}
	  \vec{V}^{\top} \vec{V}&=\brak{\norm{\vec{n}}^2\vec{I}-e^2\vec{n}\vec{n}^{\top}}^{\top}
	  \brak{\norm{\vec{n}}^2\vec{I}-e^2\vec{n}\vec{n}^{\top}}
    \\
	  \implies \vec{V}^{2} &= \norm{\vec{n}}^4\vec{I}+e^4\vec{n}\vec{n}^{\top}\vec{n}\vec{n}^{\top}
	  -2e^2\norm{\vec{n}}^2\vec{n}\vec{n}^{\top}
    \\
	  &= \norm{\vec{n}}^4\vec{I} + e^4\norm{\vec{n}}^2\vec{n}\vec{n}^{\top}
	%  \\
	  - 2e^2\norm{\vec{n}}^2\vec{n}\vec{n}^{\top}
    \\
	  &= \norm{\vec{n}}^4\vec{I} + e^2\brak{e^2 - 2}\norm{\vec{n}}^2\vec{n}\vec{n}^{\top}
    \\
	  &= \norm{\vec{n}}^4\vec{I} + \brak{e^2 - 2}\norm{\vec{n}}^2\brak{\norm{\vec{n}}^2\vec{I}- \vec{V}}
    \end{align}
%    
which can be expressed as
\begin{align}
  \vec{V}^{2} + \brak{e^2 - 2}\norm{\vec{n}}^2\vec{V} - \brak{e^2 - 1}\norm{\vec{n}}^4\vec{I}=0

\end{align}

\begin{align}
