\begin{align}
A = A \cdot 1 = A\brak{B + B'}

\begin{align}
A = AB + AB'

\solution

\noindent We can break the closed path $C$ into its four straight line paths: $O \to P \to Q \to R \to O$.

\begin{align}
\mathbf{F} \cdot d\mathbf{l} = \myvec{x + 2y \\ 2x + 4y} \cdot \myvec{dx \\ dy}
\end{align}
\begin{align}
\mathbf{F}^T d\mathbf{l} = \myvec{x + 2y \\ 2x + 4y}^T \myvec{dx \\ dy} = \myvec{x + 2y & 2x + 4y} \myvec{dx \\ dy}
\end{align}
\begin{align}
= \myvec{(x + 2y)dx + (2x + 4y)dy}
\end{align}
\noindent The general line integral expression to evaluate along any path is:
\begin{align}
\mathbf{I} = \int \brak{x + 2y}\,dx + \brak{2x + 4y}\,dy
\end{align}
\noindent{Path $O\brak{0,0} \to P\brak{1,1}$}
\begin{itemize}

    \item Limits: $x$ goes from $0$ to $1$
\end{itemize}
\begin{align}
\mathbf{I_1} = \int_0^1 \brak{x + 2x}\,dx + \brak{2x + 4x}\,dx = \int_0^1 9x\,dx = \left[ \frac{9x^2}{2} \right]_0^1 = \frac{9}{2}
\end{align}

\noindent{Path $P\brak{1,1} \to Q\brak{0,2}$}
\begin{itemize}
    \item Equation of line: $y = 2 - x \implies dy = -dx$
    \item Limits: $x$ goes from $1$ to $0$
\end{itemize}
\begin{align}
\mathbf{I_2} = \int_1^0 \brak{x + 2\brak{2-x}}\,dx + \brak{2x + 4\brak{2-x}}\brak{-dx}
\end{align}
\begin{align}
\mathbf{I_2} = \int_1^0 \left[ \brak{4-x} - \brak{8-2x} \right]\,dx = \int_1^0 \brak{x-4}\,dx = \left[ \frac{x^2}{2} - 4x \right]_1^0 = \frac{7}{2}
\end{align}

\noindent{Path $Q\brak{0,2} \to R\brak{-1,1}$}
\begin{itemize}
    \item Equation of line: $y = x + 2 \implies dy = dx$
    \item Limits: $x$ goes from $0$ to $-1$

\begin{align}
\mathbf{I_3} = \int_0^{-1} \brak{x + 2\brak{x+2}}\,dx + \brak{2x + 4\brak{x+2}}\,dx = \int_0^{-1} \brak{9x + 12}\,dx
\end{align}
\begin{align}
\mathbf{I_3} = \left[ \frac{9x^2}{2} + 12x \right]_0^{-1} = \brak{\frac{9}{2} - 12} - 0 = -\frac{15}{2}
\end{align}

\noindent{Path $R\brak{-1,1} \to O\brak{0,0}$}
\begin{itemize}

    \item Limits: $x$ goes from $-1$ to $0$

\begin{align}
\mathbf{I_4} = \int_{-1}^0 \brak{x + 2\brak{-x}}\,dx + \brak{2x + 4\brak{-x}}\brak{-dx} = \int_{-1}^0 x\,dx
\end{align}
\begin{align}
\mathbf{I_4} = \left[ \frac{x^2}{2} \right]_{-1}^0 = 0 - \frac{1}{2} = -\frac{1}{2}
\end{align}

\begin{align}

    Writing $\mathbf{u}$ and $\mathbf{v^T}$ explicitly:
    \begin{align}
    \mathbf{u}= \myvec{ u_1 \\ u_2 \\ \vdots \\ u_n }, \quad
    \mathbf{v^T} = \myvec{ v_1 & v_2 & \dots & v_n }
    \end{align}

    \noindent
    Multiplying $\mathbf{u}$ and $\mathbf{v^T}$ gives the $n \times n$ matrix $\mathbf{A}$:
    \begin{align}
    \mathbf{A} = \myvec{ 
    u_1 v_1 & u_1 v_2 & \dots & u_1 v_n \\ 
    u_2 v_1 & u_2 v_2 & \dots & u_2 v_n \\ 
    \vdots & \vdots & \ddots & \vdots \\ 
    u_n v_1 & u_n v_2 & \dots & u_n v_n 
    }
    \end{align}

    \noindent 
    Notice that each column of $\mathbf{A}$ can be written as a scalar multiple of vector $\mathbf{u}$:
    \begin{itemize}
        \item 1st column: $v_1 \mathbf{u}$
        \item 2nd column: $v_2 \mathbf{u}$
        \item $j^{\text{th}}$ column: $v_j \mathbf{u}$
    \end{itemize}
    

    \noindent 
The rank of a matrix is equal to the number of non zero rows in its echeleon form. When we reduce the above matrix into the echeleon form, we notice that each row is a multiple of another as shown above. Therefore, when we apply elementary row transformations by multiplying the rows with the necessary constants and subtracting, all rows but one become non zero. Hence, the rank of the matrix becomes $1$. \\

\solution \\

\begin{align}
\mathbf{A^T = A^{-1}}
\end{align}

\noindent Taking the determinant of both sides of the given matrix equation:
\begin{align}
\mydet{\mathbf{A^T}} = \mydet{\mathbf{A^{-1}}}
\end{align}

\noindent Using the fundamental properties of matrix determinants:
\begin{align}
         \mydet{\mathbf{A^T}} = \mydet{\mathbf{A}}
        \mydet{\mathbf{A^{-1}}} = \frac{1}{\mydet{\mathbf{A}}}
    \end{align} 

\noindent Replacing both sides using the above properties gives:
\begin{align}
\mydet{\mathbf{A}} = \frac{1}{\mydet{\mathbf{A}}}
\end{align}

\noindent Multiplying both sides by $\mydet{\mathbf{A}}$:
 \begin{align}
\brak{\mydet{\mathbf{A}}}^2 = 1 \implies \mydet{\mathbf{A}} = \pm 1
\end{align} 

\noindent \textbf{Proof that $\mydet{\mathbf{A^T}} = \mydet{\mathbf{A}}$} \\ \\
\noindent When expanding $\mydet{\mathbf{A}}$ along \textbf{Row 1}, you select entries across the top row and compute the determinants of their remaining submatrices.

\noindent When expanding $\mydet{\mathbf{A^T}}$ down \textbf{Column 1}, those exact same entries now sit vertically in the first column, and their corresponding submatrices are simply transposed.

\end{align}

Let $Y(s) = \mathcal{L}\{y\brak{x}\}$. Taking the Laplace transform on both sides:
\begin{align}
\mathcal{L}\left\{\frac{d^2 y}{dx^2}\right\} + 2\mathcal{L}\left\{\frac{dy}{dx}\right\} + \mathcal{L}\{y\} = 0
\end{align}

\noindent Using the standard operational properties:
\begin{align}
\big[s^2 Y(s) - s y\brak{0} - y'\brak{0}\big] + 2\big[s Y(s) - y\brak{0}\big] + Y(s) = 0
\end{align}

\noindent Substituting the given initial conditions $y\brak{0} = 0$ and $y'\brak{0} = 1$:

\begin{align}
(s^2 + 2s + 1) Y(s) = 1
\end{align}
\begin{align}
Y(s) = \frac{1}{(s+1)^2}
\end{align}

\item
